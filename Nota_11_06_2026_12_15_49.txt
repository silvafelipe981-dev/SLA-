def cadastrar_paciente(pacientes):

    total = len(pacientes)

    media_idade = soma_idades / total

    paciente_mais_novo = min(pacientes, key=lambda p: p["idade"])
    paciente_mais_velho = max(pacientes, key=lambda p: p["idade"])

    print(f"Total de pacientes cadastrados : {total}")

            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));

            const activeTab = document.getElementById('tab-' + tabId);

                btn.classList.remove('text-brand-600', 'bg-brand-50');
                btn.classList.add('text-slate-600');

        // Truth Table Generator
        function renderTruthTable() {
            const tbody = document.getElementById('truth-table-body');
            tbody.innerHTML = '';

            let countNormalTrue = 0;
            let countEmergenciaTrue = 0;

            let rowNum = 1;
            const bools = [true, false];

            for (let A of bools) {
                for (let B of bools) {
                    for (let C of bools) {
                        for (let D of bools) {
                            // Expressions
                            const normal = (A && B && C) || (B && C && D);
                            const emergencia = C && (B || D);

                            if (normal) countNormalTrue++;
                            if (emergencia) countEmergenciaTrue++;

                            const isTargetCase = (!A && B && C && !D);

                            const tr = document.createElement('tr');
                            tr.className = isTargetCase ? "bg-amber-100 font-bold text-amber-900 border-2 border-amber-400" : "hover:bg-slate-50";

                            tr.innerHTML = `
                                <td class="p-2 border border-slate-200">${rowNum++}</td>
                                <td class="p-2 border border-slate-200 ${A ? 'text-emerald-600':'text-rose-600'}">${A ? 'V' : 'F'}</td>
                                <td class="p-2 border border-slate-200 ${B ? 'text-emerald-600':'text-rose-600'}">${B ? 'V' : 'F'}</td>
                                <td class="p-2 border border-slate-200 ${C ? 'text-emerald-600':'text-rose-600'}">${C ? 'V' : 'F'}</td>
                                <td class="p-2 border border-slate-200 ${D ? 'text-emerald-600':'text-rose-600'}">${D ? 'V' : 'F'}</td>
                                <td class="p-2 border border-slate-200 ${normal ? 'bg-emerald-50 text-emerald-700 font-bold' : 'bg-rose-50 text-rose-700'}">${normal ? 'V (Liberado)' : 'F (Recusado)'}</td>
                                <td class="p-2 border border-slate-200 ${emergencia ? 'bg-emerald-50 text-emerald-700 font-bold' : 'bg-rose-50 text-rose-700'}">${emergencia ? 'V (Liberado)' : 'F (Recusado)'}</td>
                            `;
                            tbody.appendChild(tr);
                        }
                    }
                }
            }
        }

        // FIFO Queue Simulator Logic
        function updateQueueDisplay() {
            const container = document.getElementById('queue-display');
            container.innerHTML = '';

            if (currentQueue.length === 0) {
                container.innerHTML = `<span class="text-xs text-rose-500 font-semibold p-4">A fila de pacientes está completamente vazia!</span>`;
                return;
            }

            currentQueue.forEach((pt, index) => {
                const isFirst = index === 0;
                const card = document.createElement('div');
                card.className = `p-3 rounded-xl border flex-1 min-w-[180px] shadow-sm transition ${
                    isFirst ? 'bg-emerald-50 border-emerald-300 ring-2 ring-emerald-400/30' : 'bg-white border-slate-200'
                }`;

                card.innerHTML = `
                    <div class="flex justify-between items-center mb-1">
                        <span class="text-[10px] font-bold ${isFirst ? 'bg-emerald-200 text-emerald-800' : 'bg-slate-100 text-slate-600'} px-2 py-0.5 rounded">
                            ${isFirst ? '1º - Próximo Atendimento' : (index + 1) + 'º da Fila'}
                        </span>
                        <i class="fa-solid fa-user-clock text-xs ${isFirst ? 'text-emerald-600' : 'text-slate-400'}"></i>
                    </div>
                    <h5 class="font-bold text-xs text-slate-800">${pt.nome}</h5>
                    <p class="text-[11px] font-mono text-slate-500">CPF: ${pt.cpf}</p>
                `;
                container.appendChild(card);
            });
        }

        function dequeuePatient() {
            if (currentQueue.length === 0) return;
            const removed = currentQueue.shift();

            msgEl.className = "text-xs text-emerald-600 font-semibold";
            updateQueueDisplay();
        }
